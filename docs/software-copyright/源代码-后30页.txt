 278        const nonceStr = Formatter.nonce();

 282          Rsa.from(config.privateKey, Rsa.KEY_TYPE_PRIVATE)
 283        );

 302          notify_url: config.notifyUrl,

 323  async function verifyAndDecrypt(headers: Record<string, string>, rawBody: string) {
 324    const config = getConfig();
 325    const signature = headers["wechatpay-signature"];
 326    const timestamp = headers["wechatpay-timestamp"];
 327    const nonce = headers["wechatpay-nonce"];
 328    if (!signature || !timestamp || !nonce) throw new Error("MISSING_HEADERS");
 329    const verified = Rsa.verify(
 330      Formatter.response(timestamp, nonce, rawBody),
 331      signature,
 332      Rsa.from(config.platformPublicKey, Rsa.KEY_TYPE_PUBLIC)
 333    );
 334    if (!verified) throw new Error("VERIFY_SIGNATURE_FAILED");
 335    const body = JSON.parse(rawBody);
 336    const { resource } = body;
 337    const result = Aes.AesGcm.decrypt(
 338      resource.nonce,
 339      config.apiV3Key,
 340      resource.ciphertext,
 341      resource.associated_data
 342    ) as string;
 343    return JSON.parse(result);
 344  }
 345  export async function handlePaymentNotify(
 346    headers: Record<string, string>,
 347    rawBody: string
 348  ): Promise<{ success: boolean; message?: string }> {
 349    try {
 350      const data = await verifyAndDecrypt(headers, rawBody);
 351      console.log(`[WechatPay] 收到支付通知: ${data.out_trade_no}, 状态: ${data.trade_state}`);
 352      if (data.trade_state !== "SUCCESS") return { success: false, message: "支付未成功" };
 353      const orderNo = data.out_trade_no;
 354      const transactionId = data.transaction_id;
 355      const totalFee = data.amount.total;
 356      await prisma.$transaction(async (tx) => {
 357        const order = await tx.order.findUnique({ where: { orderNo } });
 358        if (!order) throw new Error("ORDER_NOT_FOUND");
 359        if (Math.round(Number(order.payAmount) * 100) !== totalFee) throw new Error("AMOUNT_MISMATCH");
 360        if (order.status === OrderStatus.PAID) return;
 361        await tx.order.update({
 362          where: { orderNo },
 363          data: {
 364            status: OrderStatus.PAID,
 365            paymentMethod: "wechat",
 366            paymentNo: transactionId,
 367            paymentTime: new Date(),
 368          },
 369        });
 370      });
 371      return { success: true };
 372    } catch (error) {
 373      const message = error instanceof Error ? error.message : "未知错误";
 374      console.error("[WechatPay] 处理支付通知失败:", message);
 375      return { success: false, message: message };
 376    }

 456  async function getAccessToken(): Promise<string> {
 457    const appId = process.env.WECHAT_APP_ID;
 458    const appSecret = process.env.WECHAT_APP_SECRET;
 459    if (!appId || !appSecret) {
 460      throw new Error("微信 AppID 或 AppSecret 未配置");
 461    }
 462    const now = Date.now();
 463    if (cachedAccessToken && cachedAccessToken.expiresAt > now) {
 464      return cachedAccessToken.accessToken;
 465    }
 466    const url = `https://api.weixin.qq.com/cgi-bin/token?grant_type=client_credential&appid=${appId}&secret=${appSecret}`;
 467    const response = await fetch(url);
 468    const data = await response.json();
 469    if (data.errcode) {
 470      throw new Error(`获取 Access Token 失败: ${data.errmsg}`);
 471    }
 472    cachedAccessToken = {
 473      accessToken: data.access_token,
 474      expiresAt: now + (data.expires_in - 300) * 1000,
 475    };
 476    return data.access_token;

 664    return paths.some((path) => pathname.startsWith(path));

 666  export async function middleware(request: NextRequest) {
 667    const { pathname } = request.nextUrl;
 668    const method = request.method;
 669    const token = request.cookies.get(AUTH_COOKIE_NAME)?.value;
 670    const checkAuth = async () => token ? await verifyToken(token) : false;
 671    if (matchesPath(pathname, PUBLIC_ADMIN_PATHS)) {
 672      if (token && await checkAuth()) {
 673        return NextResponse.redirect(new URL("/admin", request.url));
 674      }
 675      return NextResponse.next();
 676    }
 677    if (matchesPath(pathname, PROTECTED_PATHS)) {
 678      if (!token || !(await checkAuth())) {
 679        const loginUrl = new URL("/admin-login", request.url);
 680        loginUrl.searchParams.set("redirect", pathname);
 681        return NextResponse.redirect(loginUrl);
 682      }
 683      return NextResponse.next();
 684    }
 685    if (!pathname.startsWith("/api")) {
 686      return NextResponse.next();
 687    }
 688    if (matchesPath(pathname, PUBLIC_API_PREFIXES)) {
 689      return NextResponse.next();
 690    }
 691    if (matchesPath(pathname, SENSITIVE_API_PREFIXES)) {
 692      if (!token || !(await checkAuth())) {
 693        return jsonUnauthorized();
 694      }
 695      return NextResponse.next();
 696    }
 697    if (method !== "GET" && method !== "HEAD" && method !== "OPTIONS") {
 698      if (!token || !(await checkAuth())) {
 699        return jsonUnauthorized();
 700      }
 701    }
 702    return NextResponse.next();
 703  }
 704  function jsonUnauthorized() {
 705    return NextResponse.json(
 706      { success: false, error: { code: "UNAUTHORIZED", message: "未授权管理员访问" } },
 707      { status: 401 }
 708    );
 709  }
